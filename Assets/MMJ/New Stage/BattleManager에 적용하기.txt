{
    private StageDataSO stageData;

    private void Start()
    {
        stageData = StageRuntimeManager.Instance.CurrentStage;

        if (stageData == null)
        {
            Debug.LogError("StageDataSO가 없습니다!");
            return;
        }

        InitStage();
    }

클리어 처리 예시
public void StageClear()
{
    var stage = stageData;

    var stageSave = FindObjectOfType<StageSaveService>();
    if (stageSave == null)
    {
        Debug.LogError("StageSaveService 없음");
        return;
    }

    stageSave.ApplyClearResult(
        stage.world,
        stage.stage,
        clearTimeMs: 123456,
        score: 9999,
        stars: 3
    );

    Manager.Save.SaveCurrentUser();

    SceneManager.LoadScene("MainScene");
}
